8.5 Reviewer Score Calculation Procedure


ALTER PROC [dbo].[PRCalculateReviewerScore] @CompanyID INT
,@ReviewReviewerID INT
,@ReviewReviewerEmployeeID INT
,@EmployeeId VARCHAR(50)
AS
BEGIN
--Declare @CompanyID                INT=375, 
--@ReviewReviewerID         INT=2469,
--@ReviewReviewerEmployeeID INT =2735 
DECLARE @ResponseTypeID INT
,@ResponseTypeId2 INT
,@ResponseTypeId3 INT
,@ReviewCriteriaID INT
,@ReviewId INT
,@IsWeightedAvg BIT = 0
DECLARE @TblResponseTypeID TABLE (
id INT IDENTITY(1, 1)
,responsetypeid INT
,responsetypeid2 INT
,responseTypeId3 INT
,reviewcriteriaid DECIMAL(8, 2)
)
DECLARE @TblReviewerScore TABLE (
id INT IDENTITY(1, 1)
,reviewerscore DECIMAL(8, 2)
)

SELECT @IsWeightedAvg = weightedaverage
,@ReviewId = a.id
FROM review a
INNER JOIN reviewreviewer b ON b.reviewid = a.id
INNER JOIN review_type_score c ON c.review_id = a.id
WHERE b.id = @ReviewReviewerID

--select @IsWeightedAvg
IF @IsWeightedAvg = 1
BEGIN
INSERT INTO @TblReviewerScore (reviewerscore)
EXEC [Prcalculatereviewscorebyweightedavg] @ReviewId
,@CompanyID
,@ReviewReviewerEmployeeID
END
ELSE
BEGIN
INSERT INTO @TblResponseTypeID (
reviewcriteriaid
,responsetypeid
,responsetypeid2
,responseTypeId3
)
SELECT DISTINCT C.reviewcriteriaid
,C.responsetypeid
,C.responsetypeid2
,C.ResponseTypeId3
FROM reviewreviewer A
INNER JOIN reviewreviewercriteria B ON A.id = B.reviewreviewerid AND B.RevSUserID = @EmployeeId
INNER JOIN pr_reviewcriteria C ON C.reviewcriteriaid = B.reviewcriteriaid
LEFT OUTER JOIN jduties D ON C.description = CONVERT(VARCHAR(200), D.id)
WHERE A.id = @ReviewReviewerID and A.ReviewID = @ReviewId

DECLARE @RowCount INT
DECLARE @I INT

SET @RowCount = (
SELECT Count(*)
FROM @TblResponseTypeID
)
SET @I = 1

WHILE (@I <= @RowCount)
BEGIN
(
SELECT @ResponseTypeID = responsetypeid
,@ReviewCriteriaID = reviewcriteriaid
,@ResponseTypeId2 = responsetypeid2
,@ResponseTypeId3 = responsetypeid3
FROM @TblResponseTypeID
WHERE id = @I
)

IF (
@ResponseTypeID = 1
OR @ResponseTypeId2 = 1
OR @ResponseTypeId3 = 1
)
BEGIN
INSERT INTO @TblReviewerScore (reviewerscore)
SELECT Isnull(Avg(I.item_value), 0) + Isnull(Avg(K.item_value), 0) + Isnull(Avg(L.item_value), 0) AS ReviewerScore
FROM prreviewreviewerscoredetails A
LEFT OUTER JOIN review_score_content I ON CONVERT(VARCHAR(100), I.id) = CONVERT(VARCHAR(100), A.answers)
LEFT OUTER JOIN review_score_content K ON CONVERT(VARCHAR(100), K.id) = CONVERT(VARCHAR(100), A.answer2)
LEFT OUTER JOIN review_score_content L ON CONVERT(VARCHAR(100), L.id) = CONVERT(VARCHAR(100), A.answer3)
WHERE A.reviewreviewerid = @ReviewReviewerID
AND A.companyid = @CompanyID
AND A.reviewrevieweremployeeid = @ReviewReviewerEmployeeID
AND A.reviewcriteriaid = @ReviewCriteriaID
END

IF (
@ResponseTypeID = 3
OR @ResponseTypeID2 = 3
OR @ResponseTypeID3 = 3
)
BEGIN
INSERT INTO @TblReviewerScore (reviewerscore)
SELECT CASE 
WHEN B.responsetypeid = 3
THEN CONVERT(DECIMAL(8, 2), A.answers)
ELSE 0.0
END + CASE 
WHEN B.responsetypeid2 = 3
THEN CONVERT(DECIMAL(8, 2), A.Answer2)
ELSE 0
END + CASE 
WHEN B.responsetypeid3 = 3
THEN CONVERT(DECIMAL(8, 2), A.Answer3)
ELSE 0
END AS REVIEWSCORE
FROM prreviewreviewerscoredetails A
INNER JOIN pr_reviewcriteria B ON A.reviewcriteriaid = B.reviewcriteriaid
WHERE A.reviewreviewerid = @ReviewReviewerID
AND A.companyid = @CompanyID
AND A.reviewrevieweremployeeid = @ReviewReviewerEmployeeID
--AND B.responsetypeid = 3
AND B.reviewcriteriaid = @ReviewCriteriaID
END

SET @I = @I + 1
END
END

SELECT Sum(reviewerscore) AS ReviewerScore
FROM @TblReviewerScore
END
ALTER PROC [dbo].[PRCalculateReviewScoreByWeightedAvg] @ReviewId INT
,@CompanyID INT
,@ReviewReviewerEmployeeID INT
AS
BEGIN
--DECLARE @ReviewId int = 2571,
--@CompanyID int = 375,
--@ReviewReviewerEmployeeID int = 2735
DECLARE @ResponseTypeID INT
DECLARE @ReviewCriteriaID INT
DECLARE @ResponseTypeID2 INT
DECLARE @ResponseTypeID3 INT
DECLARE @TblResponseTypeID TABLE (
ID INT IDENTITY(1, 1)
,ReviewCriteriaID INT
,ResponseTypeID INT
,responsetypeid2 INT
,responseTypeId3 INT
)
DECLARE @TblReviewScore TABLE (
ID INT IDENTITY(1, 1)
,ReviewScore DECIMAL(18, 2)
)

INSERT INTO @TblResponseTypeID (
ReviewCriteriaID
,ResponseTypeID
,responsetypeid2
,responseTypeId3
)
SELECT C.ReviewCriteriaID
,C.ResponseTypeID
,C.ResponseTypeId2
,C.ResponseTypeId3
FROM ReviewReviewerEmployee RRE
INNER JOIN ReviewReviewer A ON RRE.ReviewReviewerID = A.ID
INNER JOIN ReviewReviewerCriteria B ON A.ID = B.ReviewReviewerID AND B.RevSUserID = RRE.UserID
INNER JOIN PR_ReviewCriteria C ON C.ReviewCriteriaID = B.ReviewCriteriaID
LEFT OUTER JOIN JDuties D ON C.Description = CONVERT(VARCHAR(200), D.ID)
WHERE A.ReviewID = @ReviewId
AND RRE.ID = @ReviewReviewerEmployeeID
GROUP BY C.ReviewCriteriaID
,C.ResponseTypeID
,C.ResponseTypeId2
,C.ResponseTypeId3

DECLARE @RowCount INT
DECLARE @I INT

SET @RowCount = (
SELECT COUNT(*)
FROM @TblResponseTypeID
)
SET @I = 1

WHILE (@I <= @RowCount)
BEGIN
(
SELECT @ResponseTypeID = ResponseTypeID
,@ReviewCriteriaID = ReviewCriteriaID
,@ResponseTypeID2 = responsetypeid2
,@ResponseTypeID3 = responseTypeId3
FROM @TblResponseTypeID
WHERE ID = @I
)

IF (@ResponseTypeID = 1
OR @ResponseTypeID2 = 1
OR @ResponseTypeID3 = 1)
BEGIN
INSERT INTO @TblReviewScore (ReviewScore)
SELECT ReviewScore = (
CASE 
WHEN G.Description = 'Accountabilities'
THEN SUM(H.Item_Value + CONVERT(DECIMAL(10, 4), ISNULL(NULLIF(J.Item_Value, ''), '0')) + CONVERT(DECIMAL(10, 4), ISNULL(NULLIF(K.Item_Value, ''), '0'))) * AVG(I.Accountability)
ELSE SUM(H.ITEM_VALUE + CONVERT(DECIMAL(10, 4), ISNULL(NULLIF(J.Item_Value, ''), '0')) + CONVERT(DECIMAL(10, 4), ISNULL(NULLIF(K.Item_Value, ''), '0'))) * AVG(I.Question)
END
) / 100
FROM ReviewReviewerEmployee A
INNER JOIN ReviewReviewerEmployee D ON D.UserID = A.UserID
AND A.ID = D.ID
INNER JOIN ReviewReviewer B ON D.ReviewReviewerID = B.ID
INNER JOIN Review C ON B.ReviewID = C.ID
INNER JOIN PRReviewReviewerScoreDetails E ON E.ReviewReviewerEmployeeID = D.ID
INNER JOIN PR_ReviewCriteria F ON F.ReviewCriteriaID = E.ReviewCriteriaID
INNER JOIN PR_ReviewCriteriaType G ON F.CriteriaTypeID = G.ReviewCriteriaTypeID
INNER JOIN Review_Score_Content H ON CONVERT(VARCHAR(max), H.ID) = E.Answers
LEFT OUTER JOIN Review_Score_Content J ON CONVERT(VARCHAR(max), J.ID) = E.Answer2
LEFT OUTER JOIN Review_Score_Content K ON CONVERT(VARCHAR(max), K.ID) = E.Answer3
INNER JOIN Review_Type_Score I ON I.Review_ID = C.ID
WHERE C.ID = @ReviewId
AND C.Company = @CompanyID
AND A.ID = @ReviewReviewerEmployeeID
AND F.ReviewCriteriaID = @ReviewCriteriaID
GROUP BY G.Description
END

IF (
@ResponseTypeID = 3
OR @ResponseTypeID2 = 3
OR @ResponseTypeID3 = 3
)
BEGIN
INSERT INTO @TblReviewScore (ReviewScore)
SELECT ReviewScore = (
CASE 
WHEN G.Description = 'Accountabilities'
THEN (
CASE 
WHEN B.responsetypeid = 3
THEN CONVERT(DECIMAL(8, 2), A.answers)
ELSE 0.0
END + CASE 
WHEN B.responsetypeid2 = 3
THEN CONVERT(DECIMAL(8, 2), A.Answer2)
ELSE 0
END + CASE 
WHEN B.responsetypeid3 = 3
THEN CONVERT(DECIMAL(8, 2), A.Answer3)
ELSE 0
END
) * AVG(I.Accountability)
ELSE (
CASE 
WHEN B.responsetypeid = 3
THEN CONVERT(DECIMAL(8, 2), A.answers)
ELSE 0.0
END + CASE 
WHEN B.responsetypeid2 = 3
THEN CONVERT(DECIMAL(8, 2), A.Answer2)
ELSE 0
END + CASE 
WHEN B.responsetypeid3 = 3
THEN CONVERT(DECIMAL(8, 2), A.Answer3)
ELSE 0
END
) * AVG(I.Question)
END
) / 100
FROM PRReviewReviewerScoreDetails A
INNER JOIN PR_ReviewCriteria B ON A.ReviewCriteriaID = B.ReviewCriteriaID
INNER JOIN PR_ReviewCriteriaType G ON B.CriteriaTypeID = G.ReviewCriteriaTypeID
INNER JOIN Review_Type_Score I ON I.Review_ID = @ReviewId
WHERE A.CompanyID = @CompanyID
AND A.ReviewReviewerEmployeeID = @ReviewReviewerEmployeeID
--AND B.ResponseTypeID = 3
AND B.ReviewCriteriaID = @ReviewCriteriaID
GROUP BY G.Description
,A.Answers
,A.Answer2
,A.Answer3
,B.responsetypeid
,B.responsetypeid2
,B.responsetypeid3
END

SET @I = @I + 1
END

SELECT Sum(ReviewScore) AS ReviewScore
FROM @TblReviewScore
END
