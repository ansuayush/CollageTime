        private static void SeedDefaults(ClientDbContext db)
        {
            if (!db.PrReviewScoreContents.Any())
            {
                db.PrReviewScoreContents.Add(new PrReviewScoreContent { ItemName = "Does Not Meet", ItemValue = 1, SortOrder = 1, IsActive = true });
                db.PrReviewScoreContents.Add(new PrReviewScoreContent { ItemName = "Needs Improvement", ItemValue = 2, SortOrder = 2, IsActive = true });
                db.PrReviewScoreContents.Add(new PrReviewScoreContent { ItemName = "Meets Expectations", ItemValue = 3, SortOrder = 3, IsActive = true });
                db.PrReviewScoreContents.Add(new PrReviewScoreContent { ItemName = "Exceeds Expectations", ItemValue = 4, SortOrder = 4, IsActive = true });
                db.PrReviewScoreContents.Add(new PrReviewScoreContent { ItemName = "Outstanding", ItemValue = 5, SortOrder = 5, IsActive = true });
            }
            if (!db.PrReviewCriteriaTypes.Any())
            {
                db.PrReviewCriteriaTypes.Add(new PrReviewCriteriaType { Description = "Accountabilities", IsActive = true });
                db.PrReviewCriteriaTypes.Add(new PrReviewCriteriaType { Description = "Competencies", IsActive = true });
                db.PrReviewCriteriaTypes.Add(new PrReviewCriteriaType { Description = "Goals", IsActive = true });
                db.PrReviewCriteriaTypes.Add(new PrReviewCriteriaType { Description = "Questions", IsActive = true });
            }
            if (!db.PrCriteriaSections.Any())
            {
                db.PrCriteriaSections.Add(new PrCriteriaSection { SectionName = "General", Description = "General performance criteria", SortOrder = 1, IsActive = true });
                db.PrCriteriaSections.Add(new PrCriteriaSection { SectionName = "Job Specific", Description = "Role-specific criteria", SortOrder = 2, IsActive = true });
            }
            db.SaveChanges();
        }
